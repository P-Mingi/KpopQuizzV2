# Step 2, review fixes: local proof of 022-photos-admin.sql on a throwaway Postgres 18 (initdb under the scratchpad). The sections of the
# first proof, plus the review's cases: the reset of the check (finding 3), the https filter (6), the closed sequences (7), the licences of
# icons and stickers (5).
set -uo pipefail
export PATH=/opt/homebrew/opt/postgresql@18/bin:$PATH LC_ALL=en_US.UTF-8 LANG=en_US.UTF-8
WT=/Users/louis/IT/Dev/projects/KpopQuizzV2/.worktrees/step2

createdb -h 127.0.0.1 -p $PORT -U postgres o1
P() { psql -h 127.0.0.1 -p $PORT -U postgres -d o1 -X -q "$@"; }
Q() { P -At -v ON_ERROR_STOP=1 -c "$1"; }
# A statement that must fail: prints the error line only.
F() { P -At -c "$1" 2>&1 | grep -E 'ERROR' | sed 's/^psql:[^E]*//'; }
# As a browser role, in a rolled back transaction; prints the rows or the error.
AS() { P -At -c "begin; set local role $1; $2; rollback;" 2>&1 | grep -v '^BEGIN$\|^ROLLBACK$\|^SET$\|^DETAIL\|^HINT' | sed 's/^psql:[^E]*//'; }
FP() { Q "select md5(concat_ws('|',
  (select string_agg(conname || pg_get_constraintdef(oid), ',' order by conname) from pg_constraint where conrelid in ('public.entity_images'::regclass, 'public.agencies'::regclass, 'public.site_assets'::regclass)),
  (select string_agg(c.relname || coalesce(c.relacl::text, '') || c.relrowsecurity::text, ',' order by c.relname) from pg_class c where c.relnamespace = 'public'::regnamespace and c.relname in ('entity_images', 'agencies', 'site_assets', 'site_assets_public', 'entity_images_public')),
  (select string_agg(attname || coalesce(attacl::text, ''), ',' order by attname) from pg_attribute where attrelid = 'public.entity_images'::regclass and attnum > 0),
  pg_get_viewdef('public.site_assets_public'::regclass), pg_get_viewdef('public.entity_images_public'::regclass),
  (select string_agg(proname || prosrc || coalesce(proacl::text, ''), ',' order by proname) from pg_proc where pronamespace = 'public'::regnamespace and proname in ('photos_touch', 'agencies_sources_ok', 'entity_images_licence_reset', 'entity_images_licence_listed')),
  (select string_agg(c.relname || coalesce(c.relacl::text, ''), ',' order by c.relname) from pg_class c where c.relkind = 'S' and c.relnamespace = 'public'::regnamespace and c.relname in ('agencies_id_seq', 'site_assets_id_seq')),
  (select string_agg(tgname, ',' order by tgname) from pg_trigger where not tgisinternal and tgrelid in ('public.entity_images'::regclass, 'public.agencies'::regclass, 'public.site_assets'::regclass)),
  (select count(*)::text from pg_policy where polrelid in ('public.agencies'::regclass, 'public.site_assets'::regclass)),
  (select string_agg(polname || pg_get_expr(polqual, polrelid), ',') from pg_policy where polrelid = 'public.entity_images'::regclass),
  (select string_agg(id || slug || name, ',' order by id) from public.agencies)))"; }
